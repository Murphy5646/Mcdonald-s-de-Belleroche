const RECRUITMENT_API_URL = "https://mcdonalds-belle-roche.wispbyte.app:11024/candidature"; const form = document.getElementById("recruitmentForm"); const result = document.getElementById("recruitmentResult"); const submitButton = form.querySelector('button[type="submit"]'); form.addEventListener("submit", async (event) => { event.preventDefault(); if (!form.reportValidity()) return; const data = new FormData(form); const candidature = { prenom: data.get("prenom").trim(), nom: data.get("nom").trim(), email: data.get("email").trim(), poste: data.get("poste").trim(), disponibilites: data.get("disponibilites").trim(), motivation: data.get("motivation").trim() }; submitButton.disabled = true; submitButton.textContent = "Envoi en cours..."; result.textContent = "Transmission de votre candidature..."; try { const response = await fetch(RECRUITMENT_API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(candidature) }); const answer = await response.json(); if (!response.ok) { throw new Error(answer.error || "Erreur lors de l'envoi."); } form.reset(); result.style.color = "#27633b"; result.textContent = "Candidature envoyée avec succès !"; } catch (error) { result.style.color = "#a33"; result.textContent = "Erreur : " + error.message; } finally { submitButton.disabled = false; submitButton.textContent = "Envoyer ma candidature"; } });
