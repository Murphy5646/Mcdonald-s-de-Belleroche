const RECRUITMENT_API_URL = "https://mcdonalds-belle-roche.wispbyte.app/candidature"; const form = document.getElementById("recruitmentForm"); const result = document.getElementById("recruitmentResult"); if (!form || !result) { console.error("Formulaire ou zone de résultat introuvable."); } else { const submitButton = form.querySelector('button[type="submit"]'); result.setAttribute("aria-live", "polite"); const setStatus = (message, type = "info") => { result.textContent = message; result.dataset.status = type; // à styliser en CSS : [data-status="error"] { color: #a33; } }; form.addEventListener("submit", async (event) => { event.preventDefault(); if (!form.reportValidity()) return; const data = new FormData(form); const candidature = Object.fromEntries( ["prenom", "nom", "email", "poste", "disponibilites", "motivation"].map( (key) => [key, String(data.get(key) || "").trim()] ) ); submitButton.disabled = true; submitButton.textContent = "Envoi en cours..."; setStatus("Transmission de votre candidature..."); const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 15000); try { const response = await fetch(RECRUITMENT_API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(candidature), signal: controller.signal, }); const answer = await response.json().catch(() => ({})); if (!response.ok) { throw new Error(answer.error || `Erreur HTTP ${response.status}`); } setStatus("Votre candidature a été envoyée avec succès !", "success"); form.reset(); } catch (error) { console.error("Erreur de candidature :", error); const message = error.name === "AbortError" ? "Le serveur met trop de temps à répondre. Réessayez dans un instant." : error instanceof TypeError ? "Impossible de contacter le serveur. Vérifiez votre connexion." : error.message; setStatus(message, "error"); } finally { clearTimeout(timeout); submitButton.disabled = false; submitButton.textContent = "Envoyer ma candidature"; } }); }
